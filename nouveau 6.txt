        private void InitializeComboBox()
        {
            // 1. Réinitialisation radicale
            cbxEtapeSuiviCmdLivre.Items.Clear();
            cbxEtapeSuiviCmdLivre.DataSource = null;
            cbxEtapeSuiviCmdLivre.DisplayMember = null;
            cbxEtapeSuiviCmdLivre.ValueMember = null;

            // 2. Chargement alternatif des données
            var etapesValides = new List<string> { "en cours", "relancée", "livrée", "réglée" };
            // Méthode A: Simple ajout manuel
            cbxEtapeSuiviCmdLivre.Items.AddRange(etapesValides.ToArray());
            // 4. Sélection par défaut
            cbxEtapeSuiviCmdLivre.SelectedIndex = 0; // Force la sélection

            // 5. Debug visuel immédiat
            MessageBox.Show($"Contenu après init: {string.Join(", ", cbxEtapeSuiviCmdLivre.Items.Cast<object>())}");
        }
		 /// <summary>
        /// Initialise le formulaire pour ajouter une nouvelle commande de livre
        /// avec l'étape de suivi par défaut = "en cours"
        /// </summary>
        private void InitNouvelleCommandeLivre()
        {
            // Tu peux aussi vider les champs si tu veux
            txbNumNewCmdLivre.Text = "";
            txbNbExemplCmdLivre.Text = "";
            txbMontantCmdLivre.Text = "";
            dateTimePickerCmdLivre.Value = DateTime.Today;
            txtbNumCmdLivreRecherche.Text = "";

            // Remplir le combo avec l'étape de suivi "en cours"
            cbxEtapeSuiviCmdLivre.Items.Clear();
            cbxEtapeSuiviCmdLivre.Items.Add("en cours");
            cbxEtapeSuiviCmdLivre.SelectedIndex = 0;
        }
